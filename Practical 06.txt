Practical 06

Aim To apply 2D geometric transformations such as translation, rotation, scaling,
reflection, and shearing on an object using matrix multiplication.


```python
import numpy as np
import matplotlib.pyplot as plt

# Original object (Triangle)
points = np.array([
    [0, 0, 1],
    [2, 0, 1],
    [1, 2, 1],
    [0, 0, 1]
])

def plot(points, title):
    plt.plot(points[:, 0], points[:, 1], 'b-o')
    plt.title(title)
    plt.xlabel("X-axis")
    plt.ylabel("Y-axis")
    plt.grid()
    plt.axis('equal')
    plt.show()

# 1. Translation
tx, ty = 2, 3
T = np.array([[1, 0, tx],
              [0, 1, ty],
              [0, 0, 1]])
translation = points @ T.T

# 2. Rotation (30 degrees)
theta = np.radians(30)
R = np.array([[np.cos(theta), -np.sin(theta), 0],
              [np.sin(theta),  np.cos(theta), 0],
              [0, 0, 1]])
rotation = points @ R.T

# 3. Scaling
S = np.array([[2, 0, 0],
              [0, 2, 0],
              [0, 0, 1]])
scaling = points @ S.T

# 4. Reflection (about X-axis)
Ref = np.array([[1, 0, 0],
                [0, -1, 0],
                [0, 0, 1]])
reflection = points @ Ref.T

# 5. Shearing (X-direction)
Sh = np.array([[1, 1, 0],
               [0, 1, 0],
               [0, 0, 1]])
shearing = points @ Sh.T

# Display all transformations
plot(points, "Original Object")
plot(translation, "Translation")
plot(rotation, "Rotation")
plot(scaling, "Scaling")
plot(reflection, "Reflection")
plot(shearing, "Shearing")
```
